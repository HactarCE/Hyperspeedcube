
use ecow::eco_format;
use hyperpuzzle_core::{
    BuildCtx, CatalogIdent, CatalogObject, Generator, GeneratorParam, GeneratorParamType,
    GeneratorSubsetParam, TypedCatalogIdValue,
};
use crate::{
    ErrorExt, EvalCtx, EvalRequestTx, FnValue, Map, Num, Result, Scope, Span, Spanned, Type, Value,
    ValueData,
};


pub(super) fn hps_obj_def_from_kwargs<T: CatalogObject>(
    id: CatalogIdent,
    kwargs: Map,
    def_span: Span,
    eval_tx: &EvalRequestTx,
    obj_from_gen_fn_output: impl 'static
    + Send
    + Sync
    + Fn(BuildCtx, &mut EvalCtx<'_>, Map) -> Result<Arc<T>>,
) -> Result<Generator<T>> {
    pop_kwarg!(kwargs, (id, id_span): String);
    let id = id.parse().at(id_span)?;

    let gen_fn = kwargs.contains_key("gen").then(|| {
        pop_kwarg!(kwargs, params: Vec<Spanned<Arc<Map>>>);
        pop_kwarg!(kwargs, (r#gen, gen_fn_span): Arc<FnValue>);
        HpsGen::Params {
            params: params_from_array(params)?,
            subset_param: None,
            gen_fn: r#gen,
            gen_fn_span,
        }
    });

    HpsObjectDef {
        id,
        kwargs: Arc::new(kwargs),
        gen_fn,
    }

    let eval_tx = eval_tx.clone();
    let obj_from_gen_fn_output = Arc::new(obj_from_gen_fn_output);

    Ok(Generator {
        id,
        params: match &hps_gen {
            HpsGen::NoParams { .. } => vec![],
            HpsGen::Params { params, .. } => params.clone(),
        },
        subset_param: None,
        validation: hyperpuzzle_core::GeneratorParamValidation { allow_empty: false },
        generate: Box::new(move |build_ctx| {
            let hps_gen = hps_gen.clone();
            let obj_from_gen_fn_output = Arc::clone(&obj_from_gen_fn_output);
            eval_tx.eval_blocking(move |runtime| {
                let mut scope = Scope::default();
                scope.special.id = Some(eco_format!("{}", build_ctx.id()));
                let scope = Arc::new(scope);

                let mut _exports = None;
                let mut ctx = EvalCtx::new(&scope, runtime, def_span, &mut _exports);

                match hps_gen {
                    HpsGen::NoParams { remaining_kwargs } => {
                        obj_from_gen_fn_output(build_ctx, &mut ctx, (*remaining_kwargs).clone())
                            .map_err(|e| runtime.report_and_convert_to_eyre(e))
                    }
                    HpsGen::Params {
                        params,
                        subset_param,
                        gen_fn,
                        gen_fn_span,
                    } => {
                        let mut params: Vec<_> =
                            std::iter::zip(&params, build_ctx.id().args.iter().cloned())
                                .map(|(p, v)| {
                                    Ok(param_value_into_hps(&p.typed_value(v).at(ctx.caller_span)?))
                                })
                                .try_collect()
                                .map_err(|e| ctx.runtime.report_and_convert_to_eyre(e))?;
                        if subset_param.is_some() {
                            params.push(match &build_ctx.id().subset {
                                Some(s) => ValueData::Str((&**s).into()).at(crate::BUILTIN_SPAN),
                                None => Value::NULL,
                            });
                        }

                        let user_gen_fn_output = gen_fn
                            .call(gen_fn_span, &mut ctx, params, Map::new())
                            .map_err(|e| ctx.runtime.report_and_convert_to_eyre(e))?;

                        match user_gen_fn_output.data {
                            ValueData::Str(redirect_id) => {
                                Ok(build_ctx.build_str_blocking(&redirect_id)?)
                            }
                            ValueData::Map(m) => {
                                obj_from_gen_fn_output(build_ctx, &mut ctx, (*m).clone())
                            }
                            _ => Err("return value of `gen` function \
                                      must be string (ID redirect) or map"
                                .at(ctx.caller_span)),
                        }
                        .map_err(|e| runtime.report_and_convert_to_eyre(e))
                    }
                }
            })
        }),
    })
}
