

            pop_kwarg!(kwargs, (id, id_span): String);
            if kwargs.contains_key("gen") {
                pop_kwarg!(kwargs, name: String = {
                    ctx.warn(eco_format!("missing `name` for puzzle generator `{id}`"));
                    id.clone()
                });
            }
            pop_kwarg!(kwargs, aliases: Vec<String> = vec![]);
            pop_kwarg!(kwargs, version: Option<String>);
            pop_kwarg!(kwargs, tags: Option<Arc<Map>>);
            pop_kwarg!(kwargs, colors: Option<String>);
            pop_kwarg!(kwargs, twists: Option<String>);
            pop_kwarg!(kwargs, (engine_name, engine_name_span): String);

            let id: CatalogIdent = id.parse().at(id_span)?;
            let engine = ctx
                .runtime
                .puzzle_engine_callback(&engine_name, engine_name_span)?;

            let g = PuzzleGeneratorInfo {
                version: super::parse_version(ctx, &format!("puzzle `{id}`"), version.as_deref())?,
                colors,
                twists,
                engine,
                generator_tags: tags_from_map(ctx, tags.unwrap_or_default()),
            };

            g.engine
                .add_additional_catalog_entries(&cat, &tx, ctx, kwargs.clone())?;

            let puzzle_engine_output_generator: Arc<Generator<HpsPuzzleEngineOutput>> =
                Arc::new(super::generators::generator_from_kwargs(
                    id.clone(),
                    kwargs,
                    ctx.caller_span,
                    &tx,
                    move |bctx, ectx, m| lazy_puzzle_from_kwargs(bctx, ectx, m, &g),
                )?);

            cat.add::<PuzzleListEntry>(Arc::new(Generator {
                id: id.clone(),
                params: puzzle_engine_output_generator.params.clone(),
                subset_param: puzzle_engine_output_generator.subset_param.clone(),
                validation: GeneratorParamValidation { allow_empty: true },
                generate: Box::new(|ctx| {
                    Ok(Arc::clone(
                        &ctx.build_blocking::<HpsPuzzleEngineOutput>(ctx.id())?.meta,
                    ))
                }),
            }))
            .at(ctx.caller_span)?;

            let tx = tx.clone();
            cat.add::<Puzzle>(Arc::new(Generator {
                id,
                params: puzzle_engine_output_generator.params.clone(),
                subset_param: puzzle_engine_output_generator.subset_param.clone(),
                validation: GeneratorParamValidation { allow_empty: false },
                generate: Box::new(move |ctx| {
                    tx.eval_blocking(|runtime| {
                        (ctx.build_blocking::<HpsPuzzleEngineOutput>(ctx.id())?
                            .build_puzzle)(ctx, runtime)
                    })
                }),
            }))
            .at(ctx.caller_span)?;

            cat.add::<HpsPuzzleEngineOutput>(puzzle_engine_output_generator)
                .at(ctx.caller_span)?;

struct PuzzleGeneratorInfo {
    version: Version,
    colors: Option<String>,
    twists: Option<String>,
    engine: Arc<dyn HpsEngine>,
    generator_tags: TagSet,
}

fn lazy_puzzle_from_kwargs(
    build_ctx: BuildCtx,
    ctx: &mut EvalCtx<'_>,
    mut kwargs: Map,
    g: &PuzzleGeneratorInfo,
) -> Result<Arc<HpsPuzzleEngineOutput>> {
    let id = build_ctx.id();
    let span = ctx.caller_span;

    pop_kwarg!(kwargs, name: String = {
        ctx.warn(eco_format!("missing `name` for puzzle `{id}`"));
        id.to_string()
    });
    pop_kwarg!(kwargs, aliases: Vec<String> = vec![]);
    pop_kwarg!(kwargs, (version, version_span): Option<String>);
    pop_kwarg!(kwargs, tags: Option<Arc<Map>>); // TODO
    let name = name;
    let aliases = aliases;
    let version: Option<Version> = version
        .map(|s| s.parse())
        .transpose()
        .map_err(|_| "error parsing version".at(version_span))?;
    let tags_map = tags;

    let mut tags = g.generator_tags.clone();
    if let Some(tags_map) = tags_map {
        tags.merge_from(tags_from_map(ctx, tags_map));
    }

    if tags.has_present("type/generator") {
        // Remove `#generator` tag.
        tags.0.remove("type/generator");

        // Add `#generated` tag.
        tags.insert_named("generated", TagValue::True)
            .at(ctx.caller_span)?;
    }

    // Add `#experimental` tag.
    tags.set_experimental_or_expect_stable(
        version.unwrap_or(Version::PLACEHOLDER),
        ctx.warnf(),
        &format!("puzzle `{id}`"),
    )
    .at(ctx.caller_span)?;
    tags.inherit_parent_tags();

    let meta = PuzzleListEntry {
        id: id.clone(),
        version,
        name,
        aliases,
        tags,
    };

    g.engine
        .make_lazy_puzzle(build_ctx, ctx, meta, kwargs)
        .at(span)
}
